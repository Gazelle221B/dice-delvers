## 全指標

## 再現

```sh
export CARGO_TARGET_DIR="$PWD/target/sim"
cargo build --release -p delvers-sim
for f in ember-frost frost-ember ember-ember frost-frost; do
  for a in greedy random; do
    for b in greedy random; do
      for preset in basic pool; do
        "$CARGO_TARGET_DIR/release/delvers-sim" 0..2000 "$a" "$b" "$f" "$preset"
      done
    done
  done
done
```
